### 1.1 S1：分层 5 折 CV 评估（主模型 L2）

来源：`outputs/scratch/S1-model.py`（`cv_evaluate` / `make_l2`）

```python
def make_l2(C=1.0, class_weight="balanced"):
    return LogisticRegression(penalty="l2", C=C, class_weight=class_weight,
                              solver="lbfgs", max_iter=1000, random_state=42)

def cv_evaluate(X, y, make_model, folds, minority):
    aucs, accs, f1s, recs = [], [], [], []
    oof_prob = np.zeros(len(y)); oof_y = np.zeros(len(y), dtype=int)
    for f in folds:
        tr = np.asarray(f["train"], dtype=int)
        te = np.asarray(f["test"], dtype=int)
        m = make_model(); m.fit(X[tr], y[tr])
        p = m.predict_proba(X[te])[:, 1]; yp = m.predict(X[te])
        oof_prob[te] = p; oof_y[te] = yp
        aucs.append(roc_auc_score(y[te], p))
        accs.append(accuracy_score(y[te], yp))
        f1s.append(f1_score(y[te], yp, pos_label=minority))
        recs.append(recall_score(y[te], yp, pos_label=minority))
    return {"AUC": float(np.mean(aucs)), "ACC": float(np.mean(accs)),
            "F1_minority": float(np.mean(f1s)),
            "Recall_minority": float(np.mean(recs))}
```

### 1.2 S2：Lasso + bootstrap 稳定性选择 + BH-FDR

来源：`outputs/scratch/S2-model.py`（`bootstrap_frequency` / `bh_qvalues`）

```python
def _fit_lasso_bootstrap(Xs, y, C, seed):
    rng = np.random.default_rng(seed)
    bd = rng.choice(np.where(y == 1)[0], size=(y == 1).sum(), replace=True)
    bh = rng.choice(np.where(y == 0)[0], size=(y == 0).sum(), replace=True)
    idx = np.concatenate([bd, bh])
    l1 = LogisticRegression(solver="liblinear", C=C, max_iter=2000,
                            random_state=seed, l1_ratio=1.0)
    l1.fit(Xs[idx], y[idx])
    return (np.abs(l1.coef_[0]) > 1e-8).astype(np.int8)

def bh_qvalues(pvals, m=None):
    pvals = np.asarray(pvals, dtype=float); n = len(pvals); m = m or n
    order = np.argsort(pvals); q = np.empty(n, dtype=float)
    q[order[-1]] = pvals[order[-1]] * m / n
    for i in range(n - 2, -1, -1):
        q[order[i]] = min(q[order[i + 1]], pvals[order[i]] * m / (i + 1))
    return q
```

### 1.3 S3：CLR 变换 + Youden 阈值 + 密度比重加权

来源：`outputs/scratch/S3-model.py`（`clr_transform` / `youden_threshold` / `_density_ratio_weights`）

```python
def clr_transform(X):
    arr = np.asarray(X, dtype=float)
    arr = np.where(arr == 0.0, CLR_DELTA, arr)   # δ=6.5e-6 乘法替换
    logX = np.log(arr)
    return logX - logX.mean(axis=1, keepdims=True)  # 逐样本几何均值中心化

def youden_threshold(y_true, y_score):
    fpr, tpr, thresholds = roc_curve(y_true, y_score)
    return float(thresholds[int(np.argmax(tpr - fpr))])  # 只在训练集估计

def _density_ratio_weights(X_clr, train_idx, test_idx, clip=10.0):
    X_all = X_clr[np.concatenate([train_idx, test_idx])]
    domain = np.concatenate([np.zeros(len(train_idx)), np.ones(len(test_idx))])
    clf = LogisticRegression(penalty="l2", C=1.0, solver="lbfgs", max_iter=2000)
    clf.fit(X_all, domain)
    w = np.exp(clf.decision_function(X_all)) * (len(train_idx) / len(test_idx))
    return np.clip(w, 0.0, clip)[:len(train_idx)]  # 转导式：用测试特征不用标签
```
