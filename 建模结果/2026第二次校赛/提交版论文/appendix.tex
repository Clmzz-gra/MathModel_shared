\subsection{核心算法}

算法 \ref{alg:s1-cv}--\ref{alg:s3-density} 给出三个子问题核心步骤的伪代码，符号与正文一致；实际实现为 Python，参数取值见正文相应小节。

\subsubsection{第一问：分层 5 折交叉验证评估}

\begin{algorithm}[H]
\caption{分层 5 折交叉验证评估（第一问，主模型 L2）}
\label{alg:s1-cv}
\begin{algorithmic}[1]
\Procedure{StratifiedCV}{$X$, $y$, model constructor, folds, minority class}
  \For{$f = 1, 2, \dots, 5$}
    \State Split train $\big(X^{(tr)}_f, y^{(tr)}_f\big)$ and test $\big(X^{(te)}_f, y^{(te)}_f\big)$ from stratified fold $f$
    \State Fit model $m_f \gets \mathrm{make\_model}\big(X^{(tr)}_f, y^{(tr)}_f\big)$
    \State Predict probabilities $p_f \gets m_f\big(X^{(te)}_f\big)$ and labels $\hat{y}_f \gets \mathbf{1}[p_f \ge 0.5]$
    \State Compute $\mathrm{AUC}_f$, $\mathrm{ACC}_f$, $\mathrm{F1}_f$, $\mathrm{Recall}_f$ with minority class as positive
  \EndFor
  \State \Return mean of each metric over the 5 folds
\EndProcedure
\end{algorithmic}
\end{algorithm}

\subsubsection{第二问：Lasso 稳定性选择与 BH-FDR 校正}

\begin{algorithm}[H]
\caption{Lasso 稀疏 Logistic 稳定性选择（第二问）}
\label{alg:s2-stable}
\begin{algorithmic}[1]
\Procedure{StabilitySelection}{$X$, $y$, $C$, $B$, $\tau$}
  \For{$b = 1, 2, \dots, B$}
    \State Draw bootstrap subset $\big(X^{(b)}, y^{(b)}\big)$ by resampling positive and negative samples with replacement
    \State Fit Lasso sparse logistic regression on $\big(X^{(b)}, y^{(b)}\big)$; obtain coefficients $\hat{\beta}^{(b)}$
    \State Record selected set $S^{(b)} = \big\{ j : |\hat{\beta}^{(b)}_j| > \varepsilon \big\}$
  \EndFor
  \State Selection frequency $\hat{\pi}_j = \tfrac{1}{B}\sum_{b=1}^{B}\mathbf{1}\big[j \in S^{(b)}\big]$
  \State Stable set $\hat{S}_\tau = \big\{ j : \hat{\pi}_j \ge \tau \big\}$ with $\tau = 0.5$
  \State \Return $\hat{S}_\tau$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{BH-FDR 多重比较校正（第二问）}
\label{alg:s2-fdr}
\begin{algorithmic}[1]
\Procedure{BH-FDR}{$p_1, p_2, \dots, p_m$}
  \State Sort in ascending order $p_{(1)} \le p_{(2)} \le \dots \le p_{(m)}$
  \State $q_{(m)} \gets p_{(m)}$
  \For{$i = m-1, m-2, \dots, 1$}
    \State $q_{(i)} \gets \min\big( q_{(i+1)},\ \tfrac{m}{i}\, p_{(i)} \big)$
  \EndFor
  \State \Return corrected q-values
\EndProcedure
\end{algorithmic}
\end{algorithm}

\subsubsection{第三问：CLR 变换、Youden 阈值与密度比重加权}

\begin{algorithm}[H]
\caption{CLR 变换与 Youden 阈值（第三问）}
\label{alg:s3-clr-youden}
\begin{algorithmic}[1]
\Procedure{CLRTransform}{$X$, $\delta$}
  \State Multiplicative replacement: $x_{ij} \leftarrow \delta = 6.5\times10^{-6}$ if $x_{ij} = 0$
  \State Center by geometric mean per sample: $\mathrm{clr}(x_{ij}) = \ln x_{ij} - \tfrac{1}{p}\sum_{k=1}^{p}\ln x_{ik}$
  \State \Return $\mathrm{clr}(X)$
\EndProcedure
\Procedure{YoudenThreshold}{$y$, $s$}
  \State Compute on training set from ROC curve $\tau^{*} = \arg\max_{\tau}\big[ \mathrm{TPR}(\tau) - \mathrm{FPR}(\tau) \big]$
  \State \Return $\tau^{*}$, estimated on training set only
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{密度比重加权（第三问，候选模型 R3）}
\label{alg:s3-density}
\begin{algorithmic}[1]
\Procedure{DensityRatioWeighting}{$X_{\mathrm{clr}}$, train indices $\mathcal{I}_{tr}$, test indices $\mathcal{I}_{te}$}
  \State Construct domain labels $d_i = 0\ (i \in \mathcal{I}_{tr})$, $d_i = 1\ (i \in \mathcal{I}_{te})$
  \State Fit domain classifier $g$: logistic regression distinguishing train/test
  \State Weights $w_i \gets \exp\big(g(x_i)\big) \cdot \tfrac{n_{tr}}{n_{te}}$, clipped to $[0, 10]$
  \State \Return training sample weights, using test features only without test labels
\EndProcedure
\end{algorithmic}
\end{algorithm}
